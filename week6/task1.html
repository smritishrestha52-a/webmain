const loadBtn = document.getElementById("loadBtn"); const saveBtn = document.getElementById("saveBtn"); const loading = document.getElementById("loading"); const errorMsg = document.getElementById("errorMsg"); const jokeSetup = document.getElementById("jokeSetup"); const jokePunch = document.getElementById("jokePunch"); const favouritesList = document.getElementById("favouritesList"); const favCount = document.getElementById("favCount"); let currentJoke = null; const backupJokes = [ { setup: "Why did the computer go to the doctor?", punchline: "Because it caught a virus!" }, { setup: "Why was the math book sad?", punchline: "Because it had too many problems." }, { setup: "Why don't eggs tell jokes?", punchline: "They might crack each other up." }, { setup: "What do you call a bear with no teeth?", punchline: "A gummy bear!" }, { setup: "Why did the student eat his homework?", punchline: "Because the teacher said it was a piece of cake!" }, { setup: "Why did the smartphone need glasses?", punchline: "It lost all its contacts!" }, { setup: "What is a computer's favourite snack?", punchline: "Microchips!" } ]; loadBtn.addEventListener("click", loadJoke); saveBtn.addEventListener("click", saveFavourite); async function loadJoke() { loading.style.display = "block"; errorMsg.style.display = "none"; loadBtn.disabled = true; saveBtn.style.display = "none"; jokeSetup.textContent = "Finding a joke..."; jokePunch.textContent = ""; currentJoke = null; try { const response = await fetch( "https://v2.jokeapi.dev/joke/Any?type=twopart&safe-mode", { cache: "no-store" } ); if (!response.ok) { throw new Error("Joke service unavailable"); } const data = await response.json(); if (data.error || !data.setup || !data.delivery) { throw new Error("Invalid joke response"); } currentJoke = { id: "api-" + Date.now(), setup: data.setup, punchline: data.delivery }; } catch (error) { console.warn("Online joke service unavailable. Using a backup joke.", error); const joke = backupJokes[ Math.floor(Math.random() * backupJokes.length) ]; currentJoke = { id: "backup-" + Date.now(), setup: joke.setup, punchline: joke.punchline }; } jokeSetup.textContent = currentJoke.setup; jokePunch.textContent = currentJoke.punchline; saveBtn.style.display = "inline-block"; loading.style.display = "none"; loadBtn.disabled = false; } function getFavourites() { try { const saved = JSON.parse( localStorage.getItem("favouriteJokes") || "[]" ); return Array.isArray(saved) ? saved : []; } catch (error) { return []; } } function saveFavourite() { if (!currentJoke) return; const favourites = getFavourites(); const exists = favourites.some( joke => joke.setup === currentJoke.setup && joke.punchline === currentJoke.punchline ); if (exists) { alert("This joke is already saved!"); return; } favourites.push(currentJoke); try { localStorage.setItem( "favouriteJokes", JSON.stringify(favourites) ); displayFavourites(); alert("Joke saved to your favourites!"); } catch (error) { alert("Could not save the joke. Please check your browser storage."); } } function displayFavourites() { const favourites = getFavourites(); favouritesList.replaceChildren(); favCount.textContent = favourites.length + " saved"; if (favourites.length === 0) { const item = document.createElement("li"); item.className = "empty"; item.textContent = "No favourites saved yet."; favouritesList.appendChild(item); return; } favourites.forEach(joke => { const item = document.createElement("li"); item.textContent = "😂 " + joke.setup + " — " + joke.punchline; favouritesList.appendChild(item); }); } displayFavourites();
